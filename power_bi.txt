Расчетные столблцы Power Query

--- Возрастная группа

= if [Возраст заемщика] <= 24 then "18–24"
  else if [Возраст заемщика] <= 34 then "25–34"
  else if [Возраст заемщика] <= 44 then "35–44"
  else if [Возраст заемщика] <= 54 then "45–54"
  else if [Возраст заемщика] <= 80 then "55+"
  else "Ошибка данных"

--- Сегмент по уровню дохода

= if [Годовой доход] < 30000 then "1. Низкий (<$30K)"
  else if [Годовой доход] < 60000 then "2. Средний ($30K–60K)"
  else if [Годовой доход] < 100000 then "3. Высокий ($60K–100K)"
  else "4. Очень высокий ($100K+)"

--- Статус кредита

= if [Статус] = 1 then "Дефолт" else "Погасил"

--- Уровень риска

= if [Кредитный грейд] = "A" or [Кредитный грейд] = "B" then "Tier 1 — Низкий риск"
  else if [Кредитный грейд] = "C" then "Tier 2 — Умеренный риск"
  else if [Кредитный грейд] = "D" then "Tier 3 — Высокий риск"
  else "Tier 4 — Критический риск"

--- 
Всего дефолтов = CALCULATE(COUNTROWS('credit_risk_dataset (2)'),'credit_risk_dataset (2)'[Статус] = 1)

--- 
Всего кредитов = COUNTROWS('credit_risk_dataset (2)') 

---
 Доля дефолтов = DIVIDE([Всего дефолтов], [Всего кредитов], 0)

---
 Метка доли дефолтов, % = FORMAT([Доля дефолтов], "0.0%")

---
 Общие потери = CALCULATE(SUM('credit_risk_dataset (2)'[Сумма займа]), 'credit_risk_dataset (2)'[Статус] = 1)

---
 Общий объем портфеля = SUM('credit_risk_dataset (2)'[Сумма займа])

---
 Средняя процентная ставка. = AVERAGE('credit_risk_dataset (2)'[Процентная ставка])

---
 Средняя сумма кредита = AVERAGE('credit_risk_dataset (2)'[Сумма займа])

---
 Сумма погашения = CALCULATE(COUNTROWS('credit_risk_dataset (2)'), 'credit_risk_dataset (2)'[Статус] = 0)

---
 Уровень потерь = DIVIDE([Общие потери], [Общий объем портфеля], 0)
